-- Sale Central: Voucher-aware checkout RPC (additive; does not replace existing billing RPC) -- Run in Supabase SQL Editor as a NEW QUERY. CREATE OR REPLACE FUNCTION public.complete_sale_with_store_credit( p_qr_codes text[], p_customer_name text DEFAULT NULL, p_customer_phone text DEFAULT NULL, p_payments jsonb DEFAULT '[]'::jsonb, p_voucher_code text DEFAULT NULL ) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $function$ DECLARE v_user_id uuid; v_item record; v_payment record; v_invoice_id uuid; v_invoice_number text; v_total numeric(12,2) := 0; v_actual_paid numeric(12,2) := 0; v_credit_used numeric(12,2) := 0; v_paid_total numeric(12,2) := 0; v_remaining_credit numeric(12,2) := 0; v_count integer := 0; v_status text; v_voucher_id uuid; v_voucher_balance numeric(12,2) := 0; v_voucher_status text; v_voucher_expires timestamptz; v_voucher_code text; BEGIN v_user_id := auth.uid(); IF v_user_id IS NULL THEN RAISE EXCEPTION 'Authentication required'; END IF; IF p_qr_codes IS NULL OR cardinality(p_qr_codes) < 1 THEN RAISE EXCEPTION 'At least one garment QR is required'; END IF; IF EXISTS (SELECT 1 FROM unnest(p_qr_codes) q(code) WHERE q.code IS NULL OR trim(q.code) = '') THEN RAISE EXCEPTION 'Invalid QR code in cart'; END IF; IF cardinality(p_qr_codes) <> (SELECT COUNT(DISTINCT q.code) FROM unnest(p_qr_codes) q(code)) THEN RAISE EXCEPTION 'Duplicate QR codes in cart'; END IF; IF p_payments IS NULL OR jsonb_typeof(p_payments) <> 'array' THEN RAISE EXCEPTION 'Payments must be a JSON array'; END IF; -- Lock and validate selected QR garments. FOR v_item IN SELECT g.id, g.product_id, g.size, g.qr_code, g.status, p.name AS product_name, p.price FROM public.garment_qr g JOIN public.products p ON p.id = g.product_id WHERE g.qr_code = ANY(p_qr_codes) AND p.user_id = v_user_id ORDER BY g.id FOR UPDATE OF g LOOP IF v_item.status <> 'in_stock' THEN RAISE EXCEPTION 'Garment QR % is not in stock', v_item.qr_code; END IF; IF v_item.price IS NULL OR v_item.price < 0 THEN RAISE EXCEPTION 'Price missing or invalid for design %', v_item.product_name; END IF; v_total := v_total + round(v_item.price, 2); v_count := v_count + 1; END LOOP; IF v_count <> cardinality(p_qr_codes) THEN RAISE EXCEPTION 'One or more QR codes are invalid or not owned by your account'; END IF; v_total := round(v_total, 2); -- Lock product rows consistently before stock changes. PERFORM p.id FROM public.products p WHERE p.user_id = v_user_id AND p.id IN (SELECT g.product_id FROM public.garment_qr g WHERE g.qr_code = ANY(p_qr_codes)) ORDER BY p.id FOR UPDATE; -- If voucher supplied, lock it and verify owner/status/expiry. IF NULLIF(trim(COALESCE(p_voucher_code, '')), '') IS NOT NULL THEN SELECT id, remaining_amount, status, expires_at, voucher_code INTO v_voucher_id, v_voucher_balance, v_voucher_status, v_voucher_expires, v_voucher_code FROM public.store_credit_vouchers WHERE user_id = v_user_id AND upper(voucher_code) = upper(trim(p_voucher_code)) FOR UPDATE; IF NOT FOUND THEN RAISE EXCEPTION 'Voucher not found'; END IF; IF v_voucher_status <> 'active' THEN RAISE EXCEPTION 'Voucher is not active'; END IF; IF v_voucher_expires IS NOT NULL AND v_voucher_expires <= now() THEN RAISE EXCEPTION 'Voucher has expired'; END IF; IF v_voucher_balance IS NULL OR v_voucher_balance <= 0 THEN RAISE EXCEPTION 'Voucher has no remaining balance'; END IF; v_credit_used := least(v_total, round(v_voucher_balance, 2)); END IF; -- Validate real tender payments. Voucher credit counts toward paid total, but is not a tender row. FOR v_payment IN SELECT value FROM jsonb_array_elements(p_payments) LOOP IF lower(trim(coalesce(v_payment.value->>'method', ''))) NOT IN ('cash', 'card', 'online') THEN RAISE EXCEPTION 'Payment method must be cash, card or online'; END IF; IF coalesce(v_payment.value->>'amount', '') !~ '^[0-9]+(\.[0-9]{1,2})?$' THEN RAISE EXCEPTION 'Invalid payment amount'; END IF; IF (v_payment.value->>'amount')::numeric <= 0 THEN RAISE EXCEPTION 'Payment amount must be greater than zero'; END IF; v_actual_paid := v_actual_paid + (v_payment.value->>'amount')::numeric; END LOOP; v_actual_paid := round(v_actual_paid, 2); IF v_actual_paid + v_credit_used > v_total THEN RAISE EXCEPTION 'Cash/card/online payments plus voucher cannot exceed invoice total'; END IF; v_paid_total := round(v_actual_paid + v_credit_used, 2); IF v_paid_total = 0 THEN v_status := 'unpaid'; ELSIF v_paid_total < v_total THEN v_status := 'partial'; ELSE v_status := 'paid'; END IF; v_invoice_id := gen_random_uuid(); v_invoice_number := 'SC-' || to_char(now(), 'YYYYMMDD') || '-' || upper(substr(replace(gen_random_uuid()::text, '-', ''), 1, 8)); INSERT INTO public.invoices (id, user_id, invoice_number, customer_name, customer_phone, subtotal, total_amount, paid_amount, balance_due, payment_status) VALUES (v_invoice_id, v_user_id, v_invoice_number, nullif(trim(coalesce(p_customer_name, '')), ''), nullif(trim(coalesce(p_customer_phone, '')), ''), v_total, v_total, v_paid_total, v_total - v_paid_total, v_status); -- Actual payment methods only (voucher has its own redemption ledger). FOR v_payment IN SELECT value FROM jsonb_array_elements(p_payments) LOOP INSERT INTO public.invoice_payments (invoice_id, payment_method, amount) VALUES (v_invoice_id, lower(trim(v_payment.value->>'method')), (v_payment.value->>'amount')::numeric); END LOOP; -- Stock deduction, sold QR state, invoice lines, and movement log. FOR v_item IN SELECT g.id, g.product_id, g.size, g.qr_code, p.name AS product_name, p.price FROM public.garment_qr g JOIN public.products p ON p.id = g.product_id WHERE g.qr_code = ANY(p_qr_codes) AND p.user_id = v_user_id ORDER BY g.id LOOP UPDATE public.product_sizes SET quantity = quantity - 1, updated_at = now() WHERE product_id = v_item.product_id AND size = v_item.size AND quantity > 0; IF NOT FOUND THEN RAISE EXCEPTION 'Insufficient stock for size % of %', v_item.size, v_item.product_name; END IF; UPDATE public.products SET quantity = quantity - 1, updated_at = now() WHERE id = v_item.product_id AND user_id = v_user_id AND quantity > 0; IF NOT FOUND THEN RAISE EXCEPTION 'Insufficient total stock for %', v_item.product_name; END IF; UPDATE public.garment_qr SET status = 'sold', sold_at = now() WHERE id = v_item.id AND status = 'in_stock'; IF NOT FOUND THEN RAISE EXCEPTION 'QR sale could not be completed'; END IF; INSERT INTO public.invoice_items (invoice_id, product_id, product_name, size, qr_code, unit_price, line_total) VALUES (v_invoice_id, v_item.product_id, v_item.product_name, v_item.size, v_item.qr_code, round(v_item.price, 2), round(v_item.price, 2)); INSERT INTO public.stock_movements (product_id, user_id, movement_type, quantity, note) VALUES (v_item.product_id, v_user_id, 'OUT', 1, 'Invoice ' || v_invoice_number || ' - QR sale - Size ' || v_item.size); END LOOP; -- Voucher debit and redemption ledger, within the same transaction. IF v_credit_used > 0 THEN v_remaining_credit := round(v_voucher_balance - v_credit_used, 2); UPDATE public.store_credit_vouchers SET remaining_amount = v_remaining_credit, status = CASE WHEN v_remaining_credit <= 0 THEN 'used' ELSE status END WHERE id = v_voucher_id AND user_id = v_user_id; IF NOT FOUND THEN RAISE EXCEPTION 'Voucher balance update failed'; END IF; INSERT INTO public.store_credit_redemptions (user_id, voucher_id, invoice_id, amount_used) VALUES (v_user_id, v_voucher_id, v_invoice_id, v_credit_used); -- Link the redemption invoice to the exchange-credit record when one exists. INSERT INTO public.exchange_history (user_id, exchange_credit_id, replacement_invoice_id, note) SELECT v_user_id, ec.id, v_invoice_id, 'Voucher redeemed: ' || v_voucher_code FROM public.exchange_credits ec WHERE ec.user_id = v_user_id AND ec.voucher_id = v_voucher_id; ELSE v_remaining_credit := v_voucher_balance; END IF; RETURN jsonb_build_object( 'success', true, 'invoice_id', v_invoice_id, 'invoice_number', v_invoice_number, 'items_count', v_count, 'total_amount', v_total, 'voucher_code', v_voucher_code, 'voucher_credit_used', v_credit_used, 'voucher_remaining', v_remaining_credit, 'paid_amount', v_paid_total, 'actual_payment_amount', v_actual_paid, 'balance_due', round(v_total - v_paid_total, 2), 'payment_status', v_status ); END; $function$; -- Restrict execution to authenticated users. (RPC is intended to be called by signed-in POS users.) REVOKE ALL ON FUNCTION public.complete_sale_with_store_credit(text[], text, text, jsonb, text) FROM PUBLIC; GRANT EXECUTE ON FUNCTION public.complete_sale_with_store_credit(text[], text, text, jsonb, text) TO authenticated;
